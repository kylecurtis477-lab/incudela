# optcov allow-list: optimizer blocks that no Forge program can reach (build-plan item 3 part c).
# `build/optcov --check` fails on a never-hit block that is not listed here, and on an entry here that the corpus
# now hits or that names no block (a stale entry). Format, one block per line:
#   function | block header (as written in selfhost/compiler.cud, stripped) | why no program reaches it
# Entries are keyed by function and header text, not line numbers, so they survive compiler edits. A new entry
# needs a reason that holds for every program the code generator can emit; a reachable hole gets a
# tests/optgate program instead (docs/tools-optcov.md).
fs_cdiv_try | if not (str_eq(o, "/") or str_eq(o, "//") or str_eq(o, "%")): | defensive: the only caller (fs_binop, level 5, operator not "*") passes /, // or %
fs_ra_slot | if not ends(o, "(%rbp)"): | defensive: callers pass only operands they matched as N(%rbp) (fs_ra_opnd, fs_ra_immq)
fs_ra_slot | if len(t) == 0: | defensive: the code generator writes every frame operand with its offset (never a bare "(%rbp)")
fs_ra_slot | if i >= len(t): | defensive: an offset of "-" alone is never emitted
fs_ra_slot | if c < 48 or c > 57: | defensive: frame offsets are decimal integers (str of an int)
fs_ra_use | if v == 2: | defensive: fs_regalloc calls fs_ra_use only after fs_ra_opnd / fs_ra_immq accepted the operand
fs_ra_use | if nsl >= 4096: | capacity: a function has at most 4096 locals (fsl_* tables), so at most 4096 distinct promotable slots; more locals stop the compiler earlier (see BUILD_PLAN item 3 note)
fs_regalloc | if not ends(l, ":\n") or contains(l, " "): | defensive: fs labels are ".LFn:" / "name:" lines; functions with asm() text are excluded before fs_regalloc runs
fs_regalloc | else: #2 | defensive: the code generator emits `rep` only as `rep movsb` / `rep stosb`
fs_regalloc | if starts(mn, "j") and not starts(ops, ".L"): | defensive: every fs jump targets an fs_newlab ".LFn" label or fs_ret (also .LF)
fs_regalloc | if nsl < 0: | follows fs_ra_use's two unreachable -1 returns (malformed operand, 4097th slot)
fs_regalloc | if nret != nlv: | defensive: every fs function body ends each exit with `leave` / `ret` (one epilogue per return)
fs_regalloc | while r < 11: | body of `if nret != nlv` (unreachable, above)
fs_ra_opnd | if len(ops) < 6: | defensive: fs_regalloc calls it only for mov/cmp lines naming %rbp, whose operands are longer
fs_ra_rsparg | if (c < 48 or c > 57) and c != 45: | defensive: the text before "(%rsp)" is always a decimal offset
fs_lv_num | if len(t) == 0: | defensive: callers pass the digits after ".LF" of a generated label, never empty
fs_lv_num | if c < 48 or c > 57: | defensive: generated labels are ".LF" + str(n)
fs_lv_build | if n == 0: | defensive: a function body always has its prologue lines when liveness runs
fs_lv_build | if not starts(ops, ".LF") or lmin < 0: | defensive: every fs jump target is an fs_newlab ".LFn" label inside the function
fs_lv_build | if v < lmin or v > lmax or lpos[v - lmin] < 0: | defensive: jumps target labels of the same function (break/continue/return/if/while labels)
fs_ra_immq | if k >= len(ops): | defensive: the operand starts with "$" and has a destination, so a comma is always present
fs_ra_1line | if char_at(l, j) == 10: | defensive: the code generator emits one instruction per out[] entry
fs_tmp_load | if k < 0: | defensive: pushed-argument reloads are written "8*k(%rsp)" with a decimal k
fs_tmp_load | if r < 0: | defensive: pushed-argument reloads before a call target argument registers only
fs_tmp_call | if n >= 6: | defensive: at most 6 arguments are reloaded into registers; further ones stay on the stack
fs_tmp_call | if pad < 0: | defensive: the alignment pad is written "sub $N,%rsp" with a decimal N
fs_tmp_call | if d < 0 or d % 8 != 0 or d // 8 >= n or tr[d // 8] >= 0: | defensive: argument reloads read distinct consecutive words above the pad
fs_tmp_call | if lr[x] == lr[q]: | defensive: each argument is reloaded into a different argument register
fs_tmp_call | if dep == 0: | defensive: argument evaluation pushes and pops %rcx in balanced pairs
fs_tmp_call | if found < n: | defensive: every reloaded argument has its `push %rax` inside the same function
fs_ssa_split | if n >= 3: | defensive: generated instructions have at most three operands (imul $N,src,dst); functions with asm() text never reach fs_regalloc
fs_ssa_line | if n < 0: | follows fs_ssa_split's unreachable -1 (more than three operands)
fs_ssa_op | if contains(substr(o, 0, b), "%") or not ends(o, ")"): | defensive: generated memory operands are disp(base[,index,scale]) without a segment prefix (segment forms come only from asm() lines)
fs_ssa_op | if r == -1 or r >= 0 and r % 2 == 1: | defensive: generated addresses use 64-bit base and index registers
fs_ssa_op | elif len(t) > 0 and fs_lv_num(t) < 0: | defensive: the scale of a generated indexed address is a decimal
fs_ssa_line | if n != 2 or starts(fs_sso[1], "$"): | defensive: generated mov/movz/movs/lea lines have a source and a register or memory destination
fs_ssa_line | if n != 2: | defensive: generated add/sub/and/or/xor lines are two-operand
fs_ssa_line | if n == 1: | defensive: the code generator writes every shift with its count ($N or %cl)
fs_ssa_multi | if not starts(t, "    ") or starts(t, "     "): | defensive: multi-line entries are only fs_ra_save's restores before `leave` and the canary set-up after the frame `sub`, all plain instructions
fs_ssa_multi | if starts(mn, "j") or str_eq(mn, "call") or str_eq(mn, "ret") or not fs_ssa_line(mn, ops, p, us, df): | defensive: those entries hold only mov/sub/leave lines
fs_ssa_multi | if a != len(l): | defensive: every out[] entry ends with a newline
fs_ssa_call | if not fs_ssa_op(ops, p, us, df, 0): | defensive: a generated call names a symbol or *%reg
fs_ssa_run | if n == 0: | defensive: a function body always has its prologue lines when the pass runs
fs_ssa_run | if lmin < 0 or v < lmin or v > lmax or lpos[v - lmin] < 0: | defensive: every fs jump target is an fs_newlab ".LFn" label of the same function (as in fs_lv_build; a jump to another label is cached as -1)
fs_ssa_row | elif fs_lv_lab(l) < 0 and not (starts(l, ".L") and ends(l, ":\n") and not contains(l, " ")): | defensive: column-0 lines inside a function body are its .LFn labels and the -g markers .Lfsd*
fs_ssa_run | if not fs_ssa_check(n, s1, s2, df, vi): | verifier: the construction always gives a well-formed SSA form; this return is only taken on a construction bug
fs_ssa_check | if x < 0: | verifier: every reached position gets a value for every register (merges set all sixteen slots at once)
fs_ssa_check | if (x & 15) != r or not df[x]: | verifier: a write id is only created for a line that writes that register
fs_ssa_check | if (x & 15) != r or vi[x - np] != x: | verifier: a phi id is only created at the position that holds it
fs_ssa_check | elif x != nw + r: | verifier: incoming-value ids are only set at the entry, per register
fs_ssa_check | if y != x and y != np + sa * 16 + r: | verifier: the merge loop runs until every edge carries the predecessor's value or the successor's phi (fall-through or jump successor)
fs_ssa_check | if y != x and y != np + sb * 16 + r: | verifier: the merge loop runs until every edge carries the predecessor's value or the successor's phi (conditional jump successor)
fs_ssa_mvdst | if cm <= sp: | defensive: the code generator emits every FS_SSMOV move with two operands
fs_ssa_rng | else: | defensive: the range after w is straight-line and w writes %rA in full, so every read of %rA there reads a value written in the range
fs_ssa_rng | if v < w * 16 or v >= (e + 1) * 16 or v % 16 != a: | defensive: e reads %rA as a copy source and the range writes %rA at w with no later unrenameable write, so the value e reads was written in the range
fs_ssa_ren | elif str_eq(g, FS_SSR32[a]): | defensive: after imul/add/sub→%rD folds, the rename corpus no longer feeds a 32-bit form of %rA that this elif would rewrite; the 64-bit path and the fs_ssa_rn refuse path stay covered
fs_ssa_ren | elif fs_ssa_rn(g) >= 0 and fs_ssa_rn(g) // 2 == a: | defensive: fs_ssa_rng already refuses a range with an %rA touch that fs_ssa_ren cannot rename
fs_ssa_ppa | if d < 0: | defensive: pad depth while walking back to the push stays non-negative (every add matches a prior pad or push)
fs_ssa_ppg | if cnt[q * 16 + 15] > 0: | defensive: the pad's flags are never read (sub $8,%rsp is only for alignment)
fs_ssa_ppg | if x < 8 or x > 8 * (d - 1): | defensive: argument loads name a slot pushed after our push (N at least 8) and not above the pad+pushes so far
fs_ssa_ppg | if q + 1 >= e or cb[q] == e or cb[q + 1] == e or not starts(out[ln[q]], "    call "): | defensive: a pad is only accepted as part of pad / loads / call / add
fs_ssa_ppg | if d < 0: | defensive: the adds return the depth the pads and inner pushes raised
fs_ssa_ppg | elif starts(l, "    pop "): | defensive: depth walk in fs_ssa_ppg sees pads/pushes/adds in the corpus; a pop between pad and matching add is not emitted
fs_ssa_ppk | if us[q * 16 + r] or (df[q * 16 + r] and cb[q] != e): | defensive: the holding register is free between the push and the pop (cold lines may write it)
fs_ssa_live | if us[p * 16 + r]: | defensive: a later read of the scratch means it is live after the pop (no kernel case needs a fresh callee-saved yet)
fs_ssa_shimm | if starts(t, "$-") or not starts(t, "$"): | defensive: every caller passes the source of a generated `mov $N,%rcx` (always `$` + decimal); negatives for shifts are emitted as `mov $N,%rax; neg; mov %rax,%rcx`, never `mov $-N,%rcx` before a %cl shift
fs_ssa_reg_rd | if starts(t, "    pop ") and (ends(t, concat(r64, "\n")) or ends(t, concat(r32, "\n"))): | defensive: fold safety looks at the first insn on each edge; a pop of the 0/1 register right after the je or at L does not appear in the corpus (the 0/1 dies at the test)
fs_ssa_reg_rd | if contains(body, r64) or contains(body, r32): | defensive: a first-edge insn that both reads and writes the 0/1 register (e.g. add %r,%r) is not the next line after the je in the corpus
fs_ssa_jcc_tgt | if q >= n: | defensive: a .LF label with no following line in the function body is not emitted
fs_ssa_fold | while fpos < n and len(out[ln[fpos]]) == 0: | defensive: the fold blanks lines only behind the scan position, so the fall-through after je is never already empty
fs_ssa_fold | if starts(t, "    set"): | defensive: add $0,%rsp is never followed by setCC in the corpus (the no-op sits after calls before mov/jmp/label)
fs_ssa_fold | if starts(t, "    j") and not starts(t, "    jmp"): | defensive: add $0,%rsp is never followed by a conditional jump in the corpus
fs_ssa_jcc_through | while q < n and (len(out[ln[q]]) == 0 or (starts(out[ln[q]], ".L") and ends(out[ln[q]], ":\n"))): | defensive: through-merge targets are single labels; stacked .L labels before the je/jne are not emitted at short-circuit merges
fs_ssa_jcc_through | if q >= n: | defensive: a .LF merge label with no following line is not emitted
fs_ssa_jcc_through | if not want_je and starts(x, "    jne "): | defensive: the corpus's through-merge cases use je L; L: je F (AND false-path); jne L; L: jne F is rare and not in the optcov programs
fs_ssa_fold | if extra: #1 | defensive: through-merge with a duplicate test before je L does not appear (dups sit on merges that keep setne, which refuse both 4x and 4ae)
fs_ssa_fold | else: #3 | defensive: the jne-L through-merge arm (FS_SSCC path) is not hit in the optcov corpus; AND false-paths use je L
fs_ssa_fold | if extra: #2 | defensive: same as if extra #1 on the through-merge success path
fs_ssa_fold | elif extra: #2 | defensive: through-merge refuse with a duplicate test to drop is not in the corpus
fs_ssa_names | if contains(s, FS_SSR[ti]) or contains(s, FS_SSR32[ti]): | defensive (step 4ai): the code generator never uses the address register of a store or load as its other operand, so a folded mov never names T again
fs_ssa_names | if FS_SSRPI[j] == ti and contains(s, FS_SSRP[j]): | defensive (step 4ai): as above, for the 16- and 8-bit names of T
fs_ssa_memfold | if fs_ssa_names(chk, ti): | defensive (step 4ai): follows fs_ssa_names; a load into T's full or 32-bit name is allowed before this test
